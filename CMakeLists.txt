cmake_minimum_required(VERSION 3.24)
project(kvmswitch VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /permissive- /sdl)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
add_library(kvm_core src/core.cpp)
target_include_directories(kvm_core PUBLIC src)
add_executable(kvmswitch src/main.cpp)
target_link_libraries(kvmswitch PRIVATE kvm_core)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Only 64-bit Windows is supported")
  endif()
  include(FetchContent)
  FetchContent_Declare(nvapi
    GIT_REPOSITORY https://github.com/NVIDIA/nvapi.git
    GIT_TAG 3d34a4faf095996663321646ebe003539a908f89)
  FetchContent_MakeAvailable(nvapi)
  target_sources(kvmswitch PRIVATE src/windows_backend.cpp)
  target_include_directories(kvmswitch PRIVATE ${nvapi_SOURCE_DIR})
  target_compile_definitions(kvmswitch PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(kvmswitch PRIVATE setupapi user32 advapi32 "${nvapi_SOURCE_DIR}/amd64/nvapi64.lib")
else()
  target_sources(kvmswitch PRIVATE src/unsupported_backend.cpp)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(command_validation_tests tests/command_validation_tests.cpp)
  target_link_libraries(command_validation_tests PRIVATE kvm_core)
  add_test(NAME command_validation COMMAND command_validation_tests)
endif()
